// ── Hook de Cuenta Regresiva ──────────────────────────────────────────────────

function useCountdown(target) {
  const [timeLeft, setTimeLeft] = useState({
    days: 0,
    hours: 0,
    minutes: 0,
    seconds: 0,
  });

  useEffect(() => {
    const tick = () => {
      const diff = target.getTime() - Date.now();
      if (diff <= 0) {
        setTimeLeft({ days: 0, hours: 0, minutes: 0, seconds: 0 });
        return;
      }
      setTimeLeft({
        days: Math.floor(diff / 86400000),
        hours: Math.floor((diff % 86400000) / 3600000),
        minutes: Math.floor((diff % 3600000) / 60000),
        seconds: Math.floor((diff % 60000) / 1000),
      });
    };
    tick();
    const id = setInterval(tick, 1000);
    return () => clearInterval(id);
  }, [target]);

  return timeLeft;
}

function Navbar() {
  const [scrolled, setScrolled] = useState(false);
  const [open, setOpen] = useState(false);

  useEffect(() => {
    const fn = () => setScrolled(window.scrollY > 48);
    window.addEventListener("scroll", fn, { passive: true });
    return () => window.removeEventListener("scroll", fn);
  }, []);

  const links = [
    { label: "Expositores", href: "#expositores" },
    { label: "Agenda", href: "#agenda" },
    { label: "Noticias", href: "#noticias" },
    { label: "Mapa", href: "#mapa" },
    { label: "Contacto", href: "#contacto" },
  ];

  return (
    <nav className={`navbar ${scrolled ? "scrolled" : ""}`}>
      <div className="container nav-content">
        <a href="#" className="nav-brand">
          <img src={expojuyLogo} alt="ExpoJuy 2026" />
        </a>

        <div className="nav-links">
          {links.map((l) => (
            <a key={l.label} href={l.href} className="nav-link">
              {l.label}
            </a>
          ))}
        </div>

        <div className="nav-actions">
          <button className="btn btn-outline">Ingresar</button>
          <button className="btn btn-primary">Comprar Entradas</button>
        </div>

        <button className="nav-toggle" onClick={() => setOpen(!open)} aria-label="Menú">
          <svg fill="none" stroke="currentColor" viewBox="0 0 24 24">
            {open ? (
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
            ) : (
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 6h16M4 12h16M4 18h16" />
            )}
          </svg>
        </button>
      </div>

      {open && (
        <div className="mobile-menu">
          {links.map((l) => (
            <a key={l.label} href={l.href} className="mobile-link" onClick={() => setOpen(false)}>
              {l.label}
            </a>
          ))}
          <button className="btn btn-primary w-full">Comprar Entradas</button>
        </div>
      )}
    </nav>
  );
}

          <div key={tier.name} className="sponsor-tier">
            <div className="tier-header">
              <div className="tier-line" style={{ background: `linear-gradient(to right, ${tier.color}40, transparent)` }} />
              <span className="tier-badge" style={{ color: tier.color, borderColor: tier.color + "30", background: tier.color + "10" }}>{tier.name}</span>
              <div className="tier-line" style={{ background: `linear-gradient(to left, ${tier.color}40, transparent)` }} />
            </div>
            <div className="sponsor-grid" style={{ gridTemplateColumns: `repeat(${tier.count}, 1fr)` }}>
              {Array.from({ length: tier.count }).map((_, i) => (
                <div key={i} className="sponsor-card" style={{ borderColor: tier.color + "1F", height: `${tier.h}px` }}>
                  <span style={{ color: tier.color + "50" }}>LOGO</span>
                </div>
              ))}
            </div>
          </div>
        ))}
        <div className="text-center mt-xl"><button className="btn btn-primary btn-large shadow-glow">Quiero ser patrocinador →</button></div>
      </div>
    </section>
  );
}

function ContactoFAQ() {
  const [openFAQ, setOpenFAQ] = useState(null);
  const [form, setForm] = useState({ name: "", email: "", message: "" });
  const [sent, setSent] = useState(false);

  return (
    <section id="contacto" className="section bg-default">
      <div className="container">
        <div className="section-header">
          <span className="section-subtitle">Contacto</span>
          <h2 className="section-title">Hablemos</h2>
        </div>
        <div className="contact-layout">
          <div className="contact-form-card">
            <h3>Contactanos</h3>
            <p className="form-desc">Para consultas sobre expositores y stands.</p>
            {sent ? (
              <div className="success-msg">
                <div className="icon">✅</div>
                <h4>Mensaje enviado</h4>
                <p>Te contactaremos en las próximas 48 horas.</p>
              </div>
            ) : (
              <form onSubmit={(e) => { e.preventDefault(); setSent(true); }} className="form">
                <div className="input-group">
                  <label>Nombre completo</label>
                  <input type="text" placeholder="Tu nombre" required value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} />
                </div>
                <div className="input-group">
                  <label>Email</label>
                  <input type="email" placeholder="tu@empresa.com" required value={form.email} onChange={(e) => setForm({ ...form, email: e.target.value })} />
                </div>
                <div className="input-group">
                  <label>Mensaje</label>
                  <textarea rows="4" placeholder="En qué podemos ayudarte..." required value={form.message} onChange={(e) => setForm({ ...form, message: e.target.value })} />
                </div>
                <button type="submit" className="btn btn-primary w-full">Enviar Mensaje</button>
              </form>
            )}
          </div>
          <div className="faq-section">
            <h3>Preguntas Frecuentes</h3>
            <div className="faq-list">
              {FAQ.map((item, i) => (
                <div key={i} className={`faq-item ${openFAQ === i ? "open" : ""}`}>
                  <button className="faq-q" onClick={() => setOpenFAQ(openFAQ === i ? null : i)}>
                    <span>{item.q}</span>
                    <svg className="faq-icon" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 4v16m8-8H4" /></svg>
                  </button>
                  {openFAQ === i && <div className="faq-a">{item.a}</div>}
                </div>
              ))}
            </div>
            <div className="faq-help">
              <div className="help-title">¿No encontrás tu respuesta?</div>
              <div className="help-desc">Escribinos a <span className="text-cyan">info@expojuy.com.ar</span></div>
            </div>
          </div>
        </div>
      </div>
    </section>
  );
}
